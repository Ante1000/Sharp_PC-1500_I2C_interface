10 REM *** PC-1500A I2C LCD 16X2 CLOCK V1.0 ***
20 REM NEEDS I2C V1.4 (NEW &4500). LCD HD44780 + PCF8574 BACKPACK (&27/&3F)
30 REM SCL = PC7 PIN 10, SDA = PB2 PIN 27. SHOWS TIME AND DATE OF THE PC-1500
40 CLEAR : DIM T$(0) * 16, P$(0) * 2 : WAIT 0
50 A0 = (PEEK &7863) * 256 : IC = A0 + &400 : IA = A0 + &4FC : TX = A0 + &300
60 IF PEEK (IC) <> &FD OR PEEK (IC + 1) <> &BE PRINT "I2C NOT INSTALLED" : END
70 D = &27 : POKE IA, D, 0, 0 : CALL IC, E : IF E = 0 GOTO 100
80 D = &3F : POKE IA, D, 0, 0 : CALL IC, E : IF E = 0 GOTO 100
90 PRINT "NO LCD AT &27 OR &3F" : BEEP 3 : END
100 PRINT "LCD FOUND, I2C ADDRESS "; D
110 REM SET THE CLOCK BEFORE RUN, IN RUN MODE: TIME = MMDDHH.MMSS
120 O = 0 : GOSUB 600 : OS = -1 : OD = -1
125 REM --- TITLE FOR 3 S, CENTERED, BOTH LINES AT ONCE ---
130 T$(0) = " * I2C CLOCK *  " : AD = &80 : O = 0 : GOSUB 700
135 T$(0) = " by Alex F4VTS  " : AD = &C0 : O = 68 : GOSUB 700
137 GOSUB 740 : O = 0
140 WAIT 192 : PRINT "I2C CLOCK BY ALEX F4VTS" : WAIT 0
200 REM --- MAIN LOOP: READ THE CLOCK, UPDATE THE LCD ---
210 T = TIME : I = INT (T) : F = INT ((T - I) * 10000 + .5) : REM MMDDHH.MMSS
220 MO = INT (I / 10000) : DA = INT ((I - MO * 10000) / 100)
225 HR = I - MO * 10000 - DA * 100
230 MI = INT (F / 100) : SE = F - MI * 100
240 IF DA = OD GOTO 270
245 N = MO : GOSUB 800 : T$(0) = "  DATE: " + P$(0) + "/"
250 N = DA : GOSUB 800 : T$(0) = T$(0) + P$(0) + "   "
255 AD = &80 : GOSUB 700 : GOSUB 740 : OD = DA
270 IF SE = OS GOTO 300
275 N = HR : GOSUB 800 : T$(0) = " TIME: " + P$(0) + ":"
280 N = MI : GOSUB 800 : T$(0) = T$(0) + P$(0) + ":"
285 N = SE : GOSUB 800 : T$(0) = T$(0) + P$(0) + " "
290 AD = &C0 : GOSUB 700 : GOSUB 740 : OS = SE
300 IF INKEY$ = "" GOTO 210
310 PRINT "STOPPED" : END
600 REM --- LCD INIT, 4-BIT MODE (BITS: D7-D4, BL=8, E=4, RW=2, RS=1) ---
605 FOR I = 1 TO 100 : NEXT I
610 FOR J = 1 TO 3 : N = &30 : GOSUB 640 : FOR I = 1 TO 30 : NEXT I : NEXT J
615 N = &20 : GOSUB 640
620 C = &28 : GOSUB 660 : C = &0C : GOSUB 660 : C = 1 : GOSUB 660
625 FOR I = 1 TO 30 : NEXT I : C = 6 : GOSUB 660 : RETURN
640 REM HIGH NIBBLE ONLY: N = NIBBLE * 16
645 POKE TX, N + 12, N + 8 : POKE IA, D, 2, 0 : CALL IC, E : RETURN
660 REM COMMAND C (RS = 0), 4 BYTES IN ONE I2C TRANSACTION
665 GOSUB 680 : POKE IA, D, 4, 0 : CALL IC, E : RETURN
680 H = C AND &F0 : L = (C AND 15) * 16
685 POKE TX + O, H + 12, H + 8, L + 12, L + 8 : RETURN
700 REM --- BUILD LINE AT TX+O: COMMAND AD (&80, &C0) + 16 CHARS OF T$(0) ---
705 C = AD : GOSUB 680 : REM TX+0..3 = ADDRESS COMMAND
710 FOR K = 0 TO 15 : C = ASC (MID$ (T$(0), K + 1, 1)) : B = O + 4 + 4 * K
715 H = C AND &F0 : L = (C AND 15) * 16
720 POKE TX + B, H + 13, H + 9, L + 13, L + 9
730 NEXT K : RETURN
740 REM SEND O + 68 BYTES (ONE OR TWO LINES) IN ONE TRANSACTION
745 POKE IA, D, O + 68, 0 : CALL IC, E : RETURN
800 REM --- TWO DIGITS OF N INTO P$(0) ---
810 Q = INT (N / 10) : P$(0) = CHR$ (48 + Q) + CHR$ (48 + N - 10 * Q) : RETURN
